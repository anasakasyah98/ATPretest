146def load_logo_emblem():
147 path = resource_path(LOGO_FILE)
148 if path is None:
149 return None
150 try:
151 from PIL import Image
152 logo = Image.open(path).convert("RGB")
153 except Exception:
154 return None
155
156 from PIL import ImageFilter, ImageOps
157
158 # Label the dark shapes on a reduced copy of the artwork
159 scale = 4
160 width, height = logo.width // scale, logo.height // scale
161 small = logo.convert("L").resize((width, height))
162 dark = [v < 160 for v in small.tobytes()]
163 labels = [0] * (width * height)
164 shapes = []
165 for start in range(width * height):
166 if not dark[start] or labels[start]:
167 continue
168 label = len(shapes) + 1
169 labels[start] = label
170 stack, pixels = [start], []
171 while stack:
172 index = stack.pop()
173 pixels.append(index)
174 x, y = index % width, index // width
175 for nx, ny in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
176 if 0 <= nx < width and 0 <= ny < height:
177 neighbor = ny * width + nx
178 if dark[neighbor] and not labels[neighbor]:
179 labels[neighbor] = label
180 stack.append(neighbor)
181 ys = [index // width for index in pixels]
182 shapes.append((pixels, min(ys), max(ys)))
183 if not shapes:
184 return None
185
186 # The text starts at the top of the first tall letter in the lower part;
187 # every shape starting from there down is text, everything above is the emblem
188 artwork_top = min(top for _, top, _ in shapes)
189 artwork_bottom = max(bottom for _, _, bottom in shapes)
190 span = artwork_bottom - artwork_top
191 letter_tops = [top for _, top, bottom in shapes
192 if top > artwork_top + 0.55 * span and bottom - top > 0.08 * span]
193 text_top = min(letter_tops, default=artwork_bottom + 1)
194 text_mask = Image.new("L", (width, height), 0)
195 text_pixels = text_mask.load()
196 emblem_box = None
197 for pixels, top, bottom in shapes:
198 if top >= text_top:
199 for index in pixels:
200 text_pixels[index % width, index // width] = 255
201 else:
202 xs = [index % width for index in pixels]
203 box = (min(xs), top, max(xs) + 1, bottom + 1)
204 emblem_box = box if emblem_box is None else (min(emblem_box[0], box[0]), min(emblem_box[1], box[1]),
205 max(emblem_box[2], box[2]), max(emblem_box[3], box[3]))
206 if emblem_box is None:
207 return None
208
209 # Erase the text, then crop the emblem into a square with a small margin
210 text_mask = text_mask.resize(logo.size, Image.NEAREST).filter(ImageFilter.MaxFilter(9))
211 logo.paste((255, 255, 255), mask=text_mask)
212 left, top, right, bottom = (value * scale for value in emblem_box)
213 side = int(max(right - left, bottom - top) * 1.04)
214 center_x, center_y = (left + right) // 2, (top + bottom) // 2
215 padded = ImageOps.expand(logo, border=side, fill=(255, 255, 255)) # Crop may reach past the edges
216 left, top = center_x - side // 2 + side, center_y - side // 2 + side
217 emblem = padded.crop((left, top, left + side, top + side))
218
219 # White background -> transparent, keeping anti-aliased edges
220 alpha = emblem.convert("L").point(lambda v: max(0, min(255, (235 - v) * 2)))
221 emblem = emblem.convert("RGBA")
222 emblem.putalpha(alpha)
223 return emblem

428def wait_for_at_ready(serial_port, timeout):
429 previous_timeout = serial_port.timeout
430 serial_port.timeout = 0.1
431 response = ""
432 try:
433 deadline = time.monotonic() + timeout
434 while time.monotonic() < deadline:
435 serial_port.reset_input_buffer()
436 serial_port.write(b"AT\r\n")
437 retry_at = min(deadline, time.monotonic() + AT_READY_RETRY_SEC)
438 while time.monotonic() < retry_at:
439 data = serial_port.read(serial_port.in_waiting or 1)
440 if data:
441 response += data.decode(errors="ignore")
442 if "OK" in response:
443 return True, response
444 return False, response
445 finally:
446 serial_port.timeout = previous_timeout

455def _probe_at_port(port):
456 try:
457 # Open Serial COM port
458 with serial.Serial(port, const.BAUD_RATE, timeout=0.2, write_timeout=1) as serial_port:
459
460 # Wait until module responds AT OK
461 is_ready, response = wait_for_at_ready(serial_port, PROBE_TIMEOUT_SEC)
462 if is_ready:
463 return True, ""
464 reply = " ".join(response.split())
465 detail = f"It replied: {reply[:60]}" if reply else "There was no reply."
466 return False, f"{port} did not answer the AT command with OK. {detail}"
467 except serial.SerialTimeoutException:
468 return False, f"{port} did not accept the AT command (write timed out)."
469 except serial.SerialException as e:
470 return False, f"Cannot open {port}: {e}"

1583 query = self.search_var.get().strip().lower()
1584
1585 # Table: matching rows only, original order kept
1586 visible = 0
1587 for number in sorted(self.records):
1588 iid = str(number)
1589 if not self.tree.exists(iid):
1590 continue
1591 record = self.records[number]
1592 text = " ".join(str(value) for value in self.tree.item(iid, "values")) + " " + record.receive
1593 if not query or query in text.lower():
1594 self.tree.move(iid, "", visible)
1595 visible += 1
1596 else:
1597 self.tree.detach(iid)

1600 self.console.tag_remove("search", "1.0", "end")

1607 position = self.console.search(query, position, "end", nocase=True, count=length)
1608 if not position:
1609 break
1610 end = f"{position}+{length.get()}c"
1611 self.console.tag_add("search", position, end)
1612 self.search_hits.append((position, end))
1613 position = end

1616 self.search_count.configure(text="", fg=COLORS["muted"])
